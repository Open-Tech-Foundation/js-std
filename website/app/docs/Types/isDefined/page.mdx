Checks if the given value is defined.

## Related

- [isUndefined](/docs/Types/isUndefined)

## Syntax

```ts
import { isDefined } from '@opentf/std';
isDefined<T>(value: T): value is Exclude<T, undefined>;
```

## Examples

```ts
isDefined() //=> false

isDefined(undefined) //=> false

isDefined(null) //=> true

isDefined('') //=> true

isDefined(0) //=> true

isDefined([]) //=> true

isDefined({}) //=> true
```

## Try it

```ts
console.log(isDefined()); //=> false

console.log(isDefined(undefined)); //=> false

console.log(isDefined(null)); //=> true

console.log(isDefined('')); //=> true

console.log(isDefined(0)); //=> true

console.log(isDefined([])); //=> true

console.log(isDefined({})); //=> true
```
